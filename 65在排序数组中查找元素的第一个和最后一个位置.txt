import java.util.*;
public class Main{
    public static void main(String[] args){
        int[] nums={5,7,7,8,8,10};
        int target=8;
        int start=-1,end=-1;
        int l=0,r=nums.length-1;
        while(l<=r){
            int mid=l+(r-l)/2;
            if(nums[mid]>target){
                r=mid-1;
            }else if(nums[mid]<target){
                l=mid+1;
            }else{
                start=mid;
                r=mid-1;//继续往左边搜
            }
        }
        l=0;
        r=nums.length-1;
        while(l<=r){
            int mid=l+(r-l)/2;
            if(nums[mid]>target){
                r=mid-1;
            }else if(nums[mid]<target){
                l=mid+1;
            }else{
                end=mid;
                l=mid+1;//继续往右边搜
            }
        }
        int[] res=new int[]{start,end};
        System.out.println(Arrays.toString(res));
    }
}
