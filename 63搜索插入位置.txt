import java.util.*;
public class Main{
    public static void main(String[] args){
        int[] nums={1,3,5,6};
        int target=5;
        int l=0,r=nums.length-1;
        while(l<=r){//区间`[left, right]`左闭右闭
            int mid=l+(r-l)/2;
            if(nums[mid]>target){
                r=mid-1;
            }else if(nums[mid]<target){
                l=mid+1;
            }else{
                System.out.println(mid);
                return;
            }
        }
        // 如果循环结束还没找到，left 就是应该插入的位置
        System.out.println(l);
    }
}
