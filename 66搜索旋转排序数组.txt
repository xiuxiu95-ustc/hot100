import java.util.*;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        System.out.println("请输入旋转排序数组(例如[4,5,6,7,0,1,2]):");
        String totalStr=sc.nextLine();
        String[] str=totalStr.substring(1,totalStr.length()-1).split(",");
        int[] nums=new int[str.length];
        for(int i=0;i<str.length;i++){
            nums[i]=Integer.parseInt(str[i]);
        }
        System.out.println("请输入目标值:");
        int target=sc.nextInt();
        int res=-1;
        int left=0;
        int right=nums.length-1;
        while(left<=right){
            //target可能不在数组内，所以只剩一个值也要走一遍循环
            int mid=left+(right-left)/2;
            if(nums[mid]==target){
                res=mid;
                break;
            }
            if(nums[left]<=nums[mid]){
                //左半段升序：检查target是否在左半段范围内
                if(nums[left]<=target && target<nums[mid]){
                    right=mid-1;
                }else{//天然处理了 target 不在数组内的情况
                    left=mid+1;
                }
            }else{
                if(nums[mid]<target && target<=nums[right]){
                    left=mid+1;
                }else{
                    right=mid-1;
                }
            }
        }
        System.out.println("result = "+res);
        sc.close();
    }
}
